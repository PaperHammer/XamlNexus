public sealed partial class __NAME__Page : ArcPage {
    public override Type ArcType => typeof(__NAME__Page);
    // Replace the sample source through new __NAME__ViewModel(yourDataSource).
    // 通过 new __NAME__ViewModel(yourDataSource) 替换示例数据源。
    public __NAME__ViewModel ViewModel { get; } = new();

    public __NAME__Page() => InitializeComponent();

    protected override async void OnEnter(FrameworkPayload? payload) {
        base.OnEnter(payload);
        string? id = payload.GetBusinessString("id");
        await ViewModel.ActivateAsync(id);
    }

    // <leave-confirmation>
    private bool confirmingLeave;
    public override async Task<bool> CanLeaveAsync(ArcLeaveReason reason, CancellationToken cancellationToken = default) {
        cancellationToken.ThrowIfCancellationRequested();
        if (confirmingLeave || ViewModel.IsBusy) return false;
        if (!ViewModel.HasUnsavedChanges) return true;
        confirmingLeave = true;
        try {
            bool chinese = LanguageUtil.CurrentLanguage?.StartsWith("zh", StringComparison.OrdinalIgnoreCase) == true;
            var dialog = new ContentDialog {
                XamlRoot = XamlRoot,
                RequestedTheme = ActualTheme,
                Title = chinese ? "有未保存修改" : "Unsaved changes",
                Content = chinese ? "可以保存并离开、放弃修改，或继续编辑。" : "Save and leave, discard your changes, or keep editing.",
                PrimaryButtonText = chinese ? "保存并离开" : "Save and leave",
                SecondaryButtonText = chinese ? "放弃修改" : "Discard changes",
                CloseButtonText = chinese ? "继续编辑" : "Keep editing",
                DefaultButton = ContentDialogButton.Close,
            };
            using var registration = cancellationToken.Register(() => DispatcherQueue.TryEnqueue(() => dialog.Hide()));
            var result = await dialog.ShowAsync();
            cancellationToken.ThrowIfCancellationRequested();
            if (result == ContentDialogResult.Secondary) return true;
            return result == ContentDialogResult.Primary && await ViewModel.TrySaveAsync();
        }
        finally { confirmingLeave = false; }
    }
    // </leave-confirmation>

    protected override async Task OnPreLeaveAsync() {
        ViewModel.Deactivate();
        await base.OnPreLeaveAsync();
    }

    protected override void OnDestroy() {
        ViewModel.Deactivate();
        base.OnDestroy();
    }

    public Visibility ToVisibility(bool value) => value ? Visibility.Visible : Visibility.Collapsed;
}

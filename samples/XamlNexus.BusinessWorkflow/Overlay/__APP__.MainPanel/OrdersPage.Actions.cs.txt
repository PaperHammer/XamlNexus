using System;
using Microsoft.UI.Xaml;
using Microsoft.UI.Xaml.Controls;
using __APP__.MainPanel.Services;
using __APP__.UIComponent.Utils;

namespace __APP__.MainPanel;

public sealed partial class OrdersPage {
    private IOrderNavigation navigation = null!;
    private async void Record_Click(object sender, ItemClickEventArgs args) {
        if (args.ClickedItem is not OrdersItem { Id: { } id }) return;
        try { await navigation.OpenAsync("OrderDetails", id); }
        catch (Exception exception) { GlobalMessageUtil.ShowException(exception); }
    }
    private async void New_Click(object sender, RoutedEventArgs args) {
        try { await navigation.OpenAsync("OrderEditor"); }
        catch (Exception exception) { GlobalMessageUtil.ShowException(exception); }
    }
}

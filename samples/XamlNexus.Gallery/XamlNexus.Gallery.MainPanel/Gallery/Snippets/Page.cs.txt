using System;
using MyApp.UIComponent.Templates;
using MyApp.UIComponent.Utils;

namespace MyApp.MainPanel;

public sealed partial class DetailsPage : ArcPage
{
    public override Type ArcType => typeof(DetailsPage);
    public DetailsPage() => InitializeComponent();

    protected override void OnEnter(FrameworkPayload? payload)
    {
        base.OnEnter(payload);
        if (payload is not null && payload.TryGet<string>("message", out var message))
            Draft.Text = message;
    }
}
